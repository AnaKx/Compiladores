program control;
var i, j, n : int;
var suma : float;

void muestra(a : int, b : float) [
    {
        if (a > 0) {
            print(a, b);
        };
    }
];

main {
    read(n);
    i = 0;
    suma = 0;
    while (i < n) do {
        j = 0;
        while (j < i) do {
            if (j > 2) {
                suma = suma + j;
            } else {
                suma = suma - 1;
            };
            j = j + 1;
        };
        i = i + 1;
    };
    muestra(i * 2, suma / n);
    if (suma < 0) {
        print("negativa");
    };
}
end
